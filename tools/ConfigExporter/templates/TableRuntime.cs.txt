            {
                Log.Error("配置加载失败，未提供根目录：__CLASS__");
                return false;

            {

                return true;

            catch (Exception exception)
            {
                Log.Error("配置文件加载失败：__CLASS__，路径：" + BytesPath, exception);
                return false;
            }
        }
